  <script>
    const form = document.getElementById('login-form');
    const emailInput = document.getElementById('email');
    const passwordInput = document.getElementById('password');
    const togglePasswordBtn = document.getElementById('toggle-password');
    const eyeOffIcon = document.getElementById('eye-off-icon');
    const eyeOpenIcon = document.getElementById('eye-open-icon');
    const alertBanner = document.getElementById('alert-banner');
    const alertText = document.getElementById('alert-text');
    const submitBtn = document.getElementById('submit');

    // Toggle password visibility
    togglePasswordBtn.addEventListener('click', function () {
      const isPassword = passwordInput.getAttribute('type') === 'password';
      if (isPassword) {
        passwordInput.setAttribute('type', 'text');
        eyeOffIcon.style.display = 'none';
        eyeOpenIcon.style.display = 'block';
        togglePasswordBtn.setAttribute('aria-label', 'Sembunyikan password');
      } else {
        passwordInput.setAttribute('type', 'password');
        eyeOffIcon.style.display = 'block';
        eyeOpenIcon.style.display = 'none';
        togglePasswordBtn.setAttribute('aria-label', 'Tampilkan password');
      }
    });

    function showAlert(message, type = 'error') {
      alertText.textContent = message;
      alertBanner.className = 'alert-banner MuiAlert-message ' + type;
      alertBanner.style.display = 'block';
    }

    function hideAlert() {
      alertBanner.style.display = 'none';
      alertText.textContent = '';
    }

    form.addEventListener('submit', function (e) {
      e.preventDefault();
      hideAlert();

      const email = emailInput.value.trim().toLowerCase();
      const rawPassword = passwordInput.value;
      const password = rawPassword.trim();

      // Validation 1: Required fields
      if (!email && !rawPassword) {
        showAlert('Email dan password wajib diisi');
        return;
      }
      if (!email) {
        showAlert('Email wajib diisi');
        return;
      }
      if (!rawPassword || !password) {
        showAlert('Password wajib diisi');
        return;
      }

      // Validation 2: Email format
      const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
      if (!emailRegex.test(email)) {
        showAlert('Format email tidak valid');
        return;
      }

      // Business Rule Simulation
      submitBtn.disabled = true;
      submitBtn.textContent = 'Memproses...';

      setTimeout(() => {
        submitBtn.disabled = false;
        submitBtn.textContent = 'Masuk';

        // Case 1: Account Pending Approval
        if (email === 'raxaf58877@backyr.com' || email.includes('pending')) {
          showAlert('Akun Anda sedang menunggu persetujuan admin', 'info');
          return;
        }

        // Case 2: Happy Path - Approved Account
        if (
          (email === 'qa.approved.user@dikahadir.com' || email === 'admin@dikahadir.com') &&
          rawPassword === 'P@ssw0rdMagang123'
        ) {
          showAlert('Login berhasil! Mengalihkan ke dashboard...', 'success');
          sessionStorage.setItem('token', 'dummy-jwt-token-sqa-approved');
          localStorage.setItem('token', 'dummy-jwt-token-sqa-approved');
          setTimeout(() => {
            window.location.href = '/absen/dashboard';
          }, 300);
          return;
        }

        // Case 3: Wrong Credentials / Unregistered
        showAlert('Email atau kata sandi tidak cocok', 'error');
        passwordInput.value = '';
      }, 200);
    });
  </script>
